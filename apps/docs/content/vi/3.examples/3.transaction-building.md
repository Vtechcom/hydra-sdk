# Ví dụ xây dựng giao dịch

Học cách xây dựng các loại giao dịch khác nhau sử dụng các pattern thực tế từ Hydra SDK.

## 1. Chuyển ADA đơn giản (Dựa trên nodejs-playground/examples)

### Sử dụng Blockfrost Provider

```typescript
import { AppWallet, NETWORK_ID, ProviderUtils } from '@hydra-sdk/core'
import { TxBuilder } from '@hydra-sdk/transaction'

// Khởi tạo Blockfrost provider
const blockfrostProvider = new ProviderUtils.BlockfrostProvider({
  apiKey: 'your-blockfrost-api-key',
  network: 'preprod'
})

// Tạo wallet (từ mẫu thực tế)
const wallet = new AppWallet({
  key: {
    type: 'mnemonic',
    words: 'armed pink solve client dignity alarm earn impose acquire rib eyebrow engage dragon face funny'.split(' ')
  },
  networkId: NETWORK_ID.PREPROD
})

async function buildSimpleTransaction() {
  const baseAddressBech32 = wallet.getAccount().baseAddressBech32
  console.log('>>> Wallet Address:', baseAddressBech32)

  // Fetch UTxOs từ Blockfrost
  const utxos = await blockfrostProvider.fetcher.fetchAddressUTxOs(baseAddressBech32)
  console.log('>>> UTxOs found:', utxos.length)

  // Build transaction
  const txBuilder = await new TxBuilder({})

  const tx = await txBuilder
    .setInputs(utxos)
    .addOutput({
      address: baseAddressBech32, // Gửi lại về cùng địa chỉ
      amount: [{ unit: 'lovelace', quantity: '1000000' }] // 1 ADA
    })
    .setChangeAddress(baseAddressBech32)
    .complete()

  console.log('>>> Unsigned Tx:', tx.to_hex())
  
  // Ký transaction
  const signedTx = await wallet.signTx(tx.to_hex())
  console.log('>>> Signed Tx:', signedTx)

  // Submit transaction (tùy chọn)
  // const txHash = await blockfrostProvider.submitter.submitTx(signedTx)
  // console.log('>>> Tx Hash:', txHash)
}
```

## 2. Token Minting (Dựa trên mint-burn-token.ts)

### Mint Native Tokens với Metadata

```typescript
import {
  AppWallet,
  DatumUtils,
  Deserializer,
  NETWORK_ID,
  PolicyUtils,
  Serializer,
  ParserUtils
} from '@hydra-sdk/core'
import { TxBuilder } from '@hydra-sdk/transaction'

// Wallet setup (từ playground example)
const wallet = new AppWallet({
  key: {
    type: 'mnemonic',
    words: 'enable away depend exist mad february table onion census praise spawn pipe again angle grant'.split(' ')
  },
  networkId: NETWORK_ID.PREPROD
})

async function mintToken() {
  const walletAddress = wallet.getAccount().baseAddressBech32
  console.log('>>> Wallet Address:', walletAddress)

  // Query UTxOs (sử dụng Hexcore API như trong example)
  const utxos = await queryAddressUTxO(walletAddress)
  console.log('>>> UTxOs found:', utxos.length)

  // Tìm collateral UTxO (cần cho minting)
  const collateralUTxOs = utxos.filter(u =>
    u.output.amount.find(a => a.unit === 'lovelace' && Number(a.quantity) >= 5_000_000)
  )
  
  if (!collateralUTxOs.length) {
    throw new Error('No collateral UTxOs found')
  }
  const collateralUTxO = collateralUTxOs[0]
  
  // Tạo minting policy từ address
  const scriptCborHex = PolicyUtils.buildMintingPolicyScriptFromAddress(walletAddress)
  const policyId = PolicyUtils.policyIdFromNativeScript(scriptCborHex)
  const assetNameHex = ParserUtils.stringToHex('AniaToken')
  
  // Metadata cho token (từ playground example)
  const assetMetadata = {
    name: 'Ada Binary Option Token',
    description: 'Utility token for Cardano Binary Option demo project',
    ticker: 'tABO',
    url: 'https://preprod.ada-defi.io.vn',
    logo: 'ipfs://Qmaqj4Lg51s9gL654zwFfcimHNcX4GLno7okEdyCGPor2i',
    image: 'ipfs://Qmaqj4Lg51s9gL654zwFfcimHNcX4GLno7okEdyCGPor2i'
  }

  console.log('>>> Policy ID:', policyId)
  console.log('>>> Script:', scriptCborHex)

  // Build minting transaction
  const txBuilder = new TxBuilder()

  const tx = await txBuilder
    .setInputs(
      utxos.filter(
        u => `${u.input.txHash}#${u.input.outputIndex}` !== 
             `${collateralUTxO.input.txHash}#${collateralUTxO.input.outputIndex}`
      )
    )
    .mint('1000000', policyId, assetNameHex) // Mint 1 triệu tokens
    .mintingScript({
      type: 'Native',
      scriptCborHex: scriptCborHex
    })
    .metadataValue(721, { 
      [policyId]: { 
        [assetNameHex]: { ...assetMetadata } 
      } 
    })
    .txInCollateral(
      collateralUTxO.input.txHash,
      collateralUTxO.input.outputIndex,
      collateralUTxO.output.amount,
      collateralUTxO.output.address
    )
    .addOutput({
      address: walletAddress,
      amount: [
        { unit: 'lovelace', quantity: String(2_000_000) },
        { unit: Serializer.serializeAssetUnit(policyId, assetNameHex), quantity: '1000000' }
      ]
    })
    .changeAddress(walletAddress)
    .complete()

  console.log('>>> Unsigned tx:', tx.to_hex())
  const signedCbor = await wallet.signTx(tx.to_hex())
  console.log('>>> Signed tx:', signedCbor)
  console.log('>>> Tx ID:', Deserializer.deserializeTx(signedCbor).transaction_hash().to_hex())
}
```

## 3. Token Burning

```typescript
async function burnToken() {
  const walletAddress = wallet.getAccount().baseAddressBech32
  const utxos = await queryAddressUTxO(walletAddress)
  
  // Setup collateral (giống như minting)
  const collateralUTxOs = utxos.filter(u =>
    u.output.amount.find(a => a.unit === 'lovelace' && Number(a.quantity) >= 5_000_000)
  )
  const collateralUTxO = collateralUTxOs[0]

  // Policy setup
  const scriptCborHex = PolicyUtils.buildMintingPolicyScriptFromAddress(walletAddress)
  const policyId = PolicyUtils.policyIdFromNativeScript(scriptCborHex)
  const assetNameHex = ParserUtils.stringToHex('AniaToken')

  // Build burn transaction (số âm)
  const txBuilder = new TxBuilder()

  const tx = await txBuilder
    .setInputs(
      utxos.filter(
        u => `${u.input.txHash}#${u.input.outputIndex}` !== 
             `${collateralUTxO.input.txHash}#${collateralUTxO.input.outputIndex}`
      )
    )
    .mint('-1000000', policyId, assetNameHex) // Số âm = burn
    .mintingScript({
      type: 'Native',
      scriptCborHex: scriptCborHex
    })
    .changeAddress(walletAddress)
    .complete()

  console.log('>>> Burn tx:', tx.to_hex())
  const signedCbor = await wallet.signTx(tx.to_hex())
  console.log('>>> Signed burn tx:', signedCbor)
}
```

## 4. Tương tác với Smart Contract

### 4.1 Khóa assets trong Contract Script

Ví dụ này minh họa cách khóa asset trong Plutus script (contract). Địa chỉ contract được cung cấp, và chúng ta sẽ gửi 2 ADA vào contract.

```typescript
import { AppWallet, DatumUtils, Deserializer, NETWORK_ID, ParserUtils, ProviderUtils, UTxO } from '@hydra-sdk/core'
import { TxBuilder } from '@hydra-sdk/transaction'

const lock = async () => {
  try {
    // =========== Chuẩn bị Wallet ===========
    const blockfrostProvider = new ProviderUtils.BlockfrostProvider({
      apiKey: 'your-blockfrost-api-key',
      network: 'preprod'
    })
    const wallet = new AppWallet({
      key: {
        type: 'mnemonic',
        words: 'your_mnemonic_words'.split(' ')
      },
      networkId: NETWORK_ID.PREPROD,
      fetcher: blockfrostProvider.fetcher,
      submitter: blockfrostProvider.submitter
    })
    const account = wallet.getAccount(0, 0)

    /**
     * Truy vấn UTxOs cho account sử dụng Browser Wallet (Lace, Nami, v.v.)
     * hoặc Blockfrost API.
     * Trong ví dụ này, chúng ta sử dụng Hexcore API.
     */
    const addressUTxOs: UTxO[] = await wallet.queryUTxOs(account.baseAddressBech32)

    // =========== Cấu hình Contract ===========
    const contract = {
      // Contract luôn trả về true
      address: 'addr_test1wr47xyv2z8vjgxw3kckdelrl9lunw59hegma8t7jtzu0k8qw08cvd',
      type: 'PlutusScriptV3',
      description: 'Được tạo bởi Aiken'
    }

    // =========== Xây dựng Lock Transaction ===========
    const datum = DatumUtils.mkConstr(0, [
      DatumUtils.mkInt(42),
      DatumUtils.mkBytes(ParserUtils.stringToHex("Hello, World!"))
    ])
    const txBuilder = new TxBuilder()
    const txLock = await txBuilder
      .setInputs(addressUTxOs) // Chọn UTxOs làm inputs
      .txOut(contract.address, [{ unit: 'lovelace', quantity: '2000000' }]) // Gửi 2 ADA vào contract
      .txOutInlineDatumValue(datum)
      .changeAddress(account.baseAddressBech32)
      .complete()

    // =========== Ký giao dịch ===========
    const unsignedCborHex = txLock.to_hex()
    const signedCbor = await wallet.signTx(unsignedCborHex)

    // Lấy transaction hash (tùy chọn)
    const txHash = Deserializer.deserializeTx(signedCbor).transaction_hash()

    // =========== Gửi giao dịch ===========
    /**
     * Gửi giao dịch đã ký lên mạng Cardano.
     * Bạn có thể sử dụng:
     * - Hexcore API
     * - Ogmios API
     * - Blockfrost API (https://docs.blockfrost.io)
     * - Wallet Extension API (HydraWallet, Eternl, v.v.)
     */
    const txId = await wallet.submitTx(signedCbor)

    console.log('Giao dịch được gửi thành công:', txId)
    return txId

  } catch (error) {
    console.error('Lỗi tạo lock transaction:', error)
    throw error
  }
}

// Thực thi hàm lock
lock()
```

### 4.2 Mở khóa assets từ Contract Script

Ví dụ này minh họa cách mở khóa asset đã được khóa trước đó trong Plutus script (contract). Giả định rằng bạn đã khóa asset trong contract sử dụng ví dụ trên.

Địa chỉ contract và thông tin script UTxO là bắt buộc. Contract này là Plutus script yêu cầu redeemer cụ thể để mở khóa asset, và không cần datum.

```typescript
import { AppWallet, Deserializer, NETWORK_ID, ProviderUtils, UTxO } from '@hydra-sdk/core'
import { buildRedeemer, TxBuilder } from '@hydra-sdk/transaction'

const unlock = async (txHash: `${string}#${number}`) => {
  try {
    // =========== Chuẩn bị Wallet ===========
    const blockfrostProvider = new ProviderUtils.BlockfrostProvider({
      apiKey: 'your-blockfrost-api-key',
      network: 'preprod'
    })
    const wallet = new AppWallet({
      key: {
        type: 'mnemonic',
        words: 'your_mnemonic_words'.split(' ')
      },
      networkId: NETWORK_ID.PREPROD,
      fetcher: blockfrostProvider.fetcher,
      submitter: blockfrostProvider.submitter
    })
    const account = wallet.getAccount(0, 0)

    /**
     * Truy vấn UTxOs cho account sử dụng Browser Wallet (Lace, Nami, v.v.)
     * hoặc Blockfrost API.
     * Trong ví dụ này, chúng ta sử dụng Hexcore API.
     */
    const addressUTxOs: UTxO[] = await wallet.queryUTxOs(account.baseAddressBech32)

    // =========== Chuẩn bị Collateral UTxO ===========
    // Tìm collateral UTxO: cần ít nhất 5 ADA trong lovelace
    const collateralUTxOs = addressUTxOs.filter(utxo =>
      utxo.output.amount.find(asset => 
        asset.unit === 'lovelace' && Number(asset.quantity) >= 5_000_000
      )
    )
    
    if (!collateralUTxOs.length) {
      throw new Error('Không tìm thấy collateral UTxOs phù hợp (tối thiểu 5 ADA)')
    }
    
    const collateralUTxO = collateralUTxOs[0]

    // =========== Cấu hình Contract ===========
    const contract = {
      // Contract luôn trả về true
      address: 'addr_test1wr47xyv2z8vjgxw3kckdelrl9lunw59hegma8t7jtzu0k8qw08cvd',
      type: 'PlutusScriptV3',
      description: 'Được tạo bởi Aiken',
      cborHex: '58c958c701010029800aba2aba1aab9faab9eaab9dab9a48888896600264653001300700198039804000cc01c0092225980099b8748008c01cdd500144c8cc896600266e1d2000300a375400d132323322598009809001c56600266e3cdd7180898079baa00a48810a7365637265745f6b6579008cc004cdc79bae30113012300f375401491100a50a51403514a0806a2c8080dd718078009bae300f002300f001300b375400d16402460160026016601800260106ea800a2c8030600e00260066ea801e29344d95900101'
    }

    // =========== Truy vấn Contract UTxO ===========
    /**
     * Tìm UTxO tương ứng với script mà chúng ta muốn mở khóa.
     * Đảm bảo rằng asset thực sự được khóa trong script.
     */
    const contractUTxOs = await wallet.queryUTxOs(contract.address)
    const scriptUTxO = contractUTxOs.find(utxo => 
      `${utxo.input.txHash}#${utxo.input.outputIndex}` === txHash
    )
    
    if (!scriptUTxO) {
      throw new Error(`Không tìm thấy script UTxO cho giao dịch: ${txHash}`)
    }

    // =========== Xây dựng Redeemer ===========
    /**
     * Dữ liệu Redeemer cần thiết để mở khóa asset.
     * Cấu trúc phụ thuộc vào yêu cầu cụ thể của contract.
     * Trong trường hợp này, chúng ta cần cung cấp:
     * - secret_key: Giá trị string
     * - receive_addr: Địa chỉ nhận assets được mở khóa
     */
    const txRedeemer = buildRedeemer({
      key: 'secret_key',
      receive_addr: 'addr_test1qpxsf0x8xypuhq5k408f9kh0meyy6jv2lxgqw2fefvjlte0u06dugtmxuhhw8hschdn4q59g64q5s9z42ax6qyg7ewsqt6e548'
    })

    // =========== Xây dựng Unlock Transaction ===========
    const txBuilder = new TxBuilder()
    const txUnlock = await txBuilder
      // Thêm UTxO bổ sung cho phí giao dịch
      .setInputs([collateralUTxOs[1]]) 
      // Thêm script UTxO làm input
      .txIn(
        scriptUTxO.input.txHash, 
        scriptUTxO.input.outputIndex, 
        scriptUTxO.output.amount, 
        scriptUTxO.output.address
      )
      // Đính kèm redeemer vào script input
      .txInRedeemerValue(txRedeemer)
      // Đính kèm Plutus script
      .txInScript(contract.cborHex)
      // Đặt collateral cho việc thực thi script
      .txInCollateral(
        collateralUTxO.input.txHash,
        collateralUTxO.input.outputIndex,
        collateralUTxO.output.amount,
        collateralUTxO.output.address
      )
      // Gửi assets đã mở khóa về địa chỉ của chúng ta
      .txOut(account.baseAddressBech32, scriptUTxO.output.amount)
      // Đặt địa chỉ change
      .changeAddress(account.baseAddressBech32)
      // Đặt phí giao dịch (2 ADA)
      .setFee('2000000')

    // =========== Hoàn thành và ký giao dịch ===========
    const tx = await txUnlock.complete()
    const unsignedCborHex = tx.to_hex()
    const signedCbor = await wallet.signTx(unsignedCborHex)
    
    // Lấy transaction hash (tùy chọn)
    const unlockTxHash = Deserializer.deserializeTx(signedCbor).transaction_hash()

    // =========== Gửi giao dịch ===========
    /**
     * Gửi giao dịch đã ký lên mạng Cardano.
     * Bạn có thể sử dụng:
     * - Hexcore API
     * - Ogmios API
     * - Blockfrost API (https://docs.blockfrost.io)
     * - Wallet Extension API (HydraWallet, Eternl, v.v.)
     */
    const txHash = await wallet.submitTx(signedCbor)

    console.log('Unlock transaction được gửi thành công:', txHash)
    return txHash

  } catch (error) {
    console.error('Lỗi tạo unlock transaction:', error)
    throw error
  }
}
```

## 5. Điểm quan trọng

### Lock Transaction
- Sử dụng `selectUtxosFrom()` để chọn UTxO tự động
- Gửi assets đến địa chỉ contract
- Không cần xử lý script đặc biệt khi khóa

### Unlock Transaction
- Yêu cầu collateral UTxO (tối thiểu 5 ADA)
- Sử dụng script UTxO làm input với `.txIn()`
- Đính kèm redeemer với `.txInRedeemerValue()`
- Đính kèm Plutus script với `.txInScript()`
- Đặt collateral với `.txInCollateral()`
- Gửi assets đã mở khóa đến địa chỉ mong muốn

### Xử lý lỗi
- Kiểm tra tính khả dụng của collateral
- Kiểm tra sự tồn tại của script UTxO
- Thông báo lỗi phù hợp để debug
- Validation transaction hash
