# Ví dụ Tạo Ví

Học cách tạo và quản lý ví bằng Hydra SDK trên các nền tảng và framework khác nhau.

## Ví dụ Node.js

Tạo ví server-side với Node.js:

```javascript
// wallet-manager.js
const { AppWallet, generateMnemonic, NETWORK_ID } = require('@hydra-sdk/core');

class WalletManager {
    constructor(networkId = NETWORK_ID.TESTNET) {
        this.networkId = networkId;
    }

    createNewWallet() {
        console.log('Đang tạo ví mới...');
        
        // Tạo mnemonic mới
        const mnemonic = generateMnemonic();
        console.log('Mnemonic đã tạo:', mnemonic);
        
        // Tạo ví
        const wallet = new AppWallet({
            key: {
                type: 'mnemonic',
                words: mnemonic.split(' ')
            },
            networkId: this.networkId
        });

        return {
            wallet,
            mnemonic,
            account: wallet.getAccount(0, 0)
        };
    }

    restoreWallet(mnemonic) {
        console.log('Đang khôi phục ví từ mnemonic...');
        
        const wallet = new AppWallet({
            key: {
                type: 'mnemonic',
                words: mnemonic.split(' ')
            },
            networkId: this.networkId
        });

        return {
            wallet,
            mnemonic,
            account: wallet.getAccount(0, 0)
        };
    }

    getWalletInfo(walletData) {
        const { account, mnemonic } = walletData;
        
        return {
            mnemonic,
            baseAddress: account.baseAddressBech32,
            stakeAddress: account.stakeAddressBech32,
            paymentKeyHash: account.paymentKeyHash,
            stakeKeyHash: account.stakeKeyHash
        };
    }
}

// Ví dụ sử dụng
async function main() {
    const walletManager = new WalletManager(NETWORK_ID.TESTNET);
    
    // Tạo ví mới
    const newWallet = walletManager.createNewWallet();
    console.log('Thông tin ví mới:', walletManager.getWalletInfo(newWallet));
    
    // Khôi phục ví hiện có
    const existingMnemonic = 'abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon abandon art';
    const restoredWallet = walletManager.restoreWallet(existingMnemonic);
    console.log('Thông tin ví đã khôi phục:', walletManager.getWalletInfo(restoredWallet));
}

main().catch(console.error);

module.exports = { WalletManager };
```

<script setup lang="ts">
import { ref, reactive } from 'vue'

interface WalletInfo {
  mnemonic: string
  baseAddress: string
  stakeAddress: string
  paymentKeyHash: string
}

const walletInfo = ref<WalletInfo | null>(null)
const loading = ref(false)
const mnemonicInput = ref('')
const networkId = ref(NETWORK_ID.TESTNET)

  loading.value = true
  
  try {
    // Tạo mnemonic mới

  walletInfo.value = null

// WalletCreation.tsx
import React, { useState } from 'react';
import { AppWallet, generateMnemonic, NETWORK_ID } from '@hydra-sdk/core';

interface WalletInfo {
  mnemonic: string;
  baseAddress: string;
  stakeAddress: string;
  paymentKeyHash: string;
}

const WalletCreation: React.FC = () => {
  const [walletInfo, setWalletInfo] = useState<WalletInfo | null>(null);
  const [loading, setLoading] = useState(false);
  const [mnemonicInput, setMnemonicInput] = useState('');
  const [networkId, setNetworkId] = useState(NETWORK_ID.TESTNET);

  const createNewWallet = async () => {
    setLoading(true);
    
    try {
      // Tạo mnemonic mới
      const mnemonic = generateMnemonic();
      
      // Tạo ví
      const wallet = new AppWallet({
        key: {
          type: 'mnemonic',
          words: mnemonic.split(' ')
        },
        networkId
      });

      const account = wallet.getAccount(0, 0);
      
      setWalletInfo({
        mnemonic,
        baseAddress: account.baseAddressBech32,
        stakeAddress: account.stakeAddressBech32,
        paymentKeyHash: account.paymentKeyHash
      });
    } catch (error) {
      console.error('Lỗi khi tạo ví:', error);
    } finally {
      setLoading(false);
    }
  };

  const restoreWallet = async () => {
    if (!mnemonicInput.trim()) {
      alert('Vui lòng nhập cụm từ mnemonic hợp lệ');
      return;
    }

    setLoading(true);
    
    try {
      const wallet = new AppWallet({
        key: {
          type: 'mnemonic',
          words: mnemonicInput.trim().split(' ')
        },
        networkId
      });

      const account = wallet.getAccount(0, 0);
      
      setWalletInfo({
        mnemonic: mnemonicInput.trim(),
        baseAddress: account.baseAddressBech32,
        stakeAddress: account.stakeAddressBech32,
        paymentKeyHash: account.paymentKeyHash
      });
    } catch (error) {
      console.error('Lỗi khi khôi phục ví:', error);
      alert('Lỗi khôi phục ví. Vui lòng kiểm tra cụm từ mnemonic.');

      setLoading(false);

    setWalletInfo(null);
    setMnemonicInput('');

    <div className="wallet-container">
      <h1>Tạo Ví Hydra</h1>

          <div className="info-item">

    </div>
  );
};

export default WalletCreation;

<script lang="ts">
  import { onMount } from 'svelte';
  import { writable } from 'svelte/store';
  import { AppWallet, generateMnemonic, NETWORK_ID } from '@hydra-sdk/core';

  interface WalletInfo {
    mnemonic: string;
    baseAddress: string;
    stakeAddress: string;
    paymentKeyHash: string;
  }

  let walletInfo: WalletInfo | null = null;
  let loading = false;
  let mnemonicInput = '';
  let networkId = NETWORK_ID.TESTNET;

  // Svelte store cho trạng thái ví
  const walletStore = writable<WalletInfo | null>(null);

  onMount(() => {
    // Tải ví từ localStorage
    const stored = localStorage.getItem('hydra-wallet');
    if (stored) {
      walletInfo = JSON.parse(stored);
      walletStore.set(walletInfo);
    }
  });

  // Đăng ký theo dõi thay đổi wallet store
  walletStore.subscribe(value => {
    if (value) {
      localStorage.setItem('hydra-wallet', JSON.stringify(value));
    }
  });

  async function createNewWallet() {
    loading = true;
    
    try {
      const mnemonic = generateMnemonic();
      
      const wallet = new AppWallet({
        key: {
          type: 'mnemonic',
          words: mnemonic.split(' ')
        },
        networkId
      });

      const account = wallet.getAccount(0, 0);
      
      walletInfo = {
        mnemonic,
        baseAddress: account.baseAddressBech32,
        stakeAddress: account.stakeAddressBech32,
        paymentKeyHash: account.paymentKeyHash
      };

      walletStore.set(walletInfo);
    } catch (error) {
      console.error('Lỗi khi tạo ví:', error);
      alert('Không thể tạo ví');
    } finally {
      loading = false;
    }
  }

  async function restoreWallet() {
    if (!mnemonicInput.trim()) {
      alert('Vui lòng nhập cụm từ mnemonic hợp lệ');
      return;

    loading = true;

    } finally {
      loading = false;
    }
  }

  function clearWallet() {
    walletInfo = null;
    mnemonicInput = '';
    localStorage.removeItem('hydra-wallet');
    walletStore.set(null);
  }

## Các Tính Năng Chính

Tất cả các ví dụ đều minh họa:

- **Tạo Mnemonic**: Tạo cụm từ mnemonic 24 từ mới
- **Khôi Phục Ví**: Khôi phục ví từ mnemonic có sẵn  
- **Lựa Chọn Mạng**: Chọn giữa Testnet và Mainnet
- **Tạo Địa Chỉ**: Lấy base address và stake address
- **Quản Lý Key**: Truy cập payment và stake key hash
- **Lưu Trữ**: Lưu dữ liệu ví vào localStorage (khi có thể)
- **Xử Lý Lỗi**: Quản lý lỗi và phản hồi người dùng đúng cách
- **Trạng Thái Loading**: Phản hồi UI trong quá trình xử lý bất đồng bộ

Mỗi ví dụ đặc thù cho framework và tuân theo best practices của framework tương ứng về quản lý state, xử lý lifecycle và kiến trúc component.
