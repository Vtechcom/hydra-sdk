## Complex Datum Creation (Từ mint-burn-token.ts)

### Multi-level Constructor Datum

```typescript
import { DatumUtils, ParserUtils } from '@hydra-sdk/core'
import { CardanoWASM } from '@hydra-sdk/cardano-wasm'

// Complex datum builder từ thực tế trong playground
const buildDatum = (key: string, l1Vkh: string, l2Vkh: string, amount: string) => {
  // Inner constructors
  const bKey = DatumUtils.mkBytes(key)
  const cL1Vkh = DatumUtils.mkConstr(0, [DatumUtils.mkBytes(l1Vkh)])
  const cL2Vkh = DatumUtils.mkConstr(0, [DatumUtils.mkBytes(l2Vkh)])

  // Complex nested constructor
  const constrKey = DatumUtils.mkConstr(0, [bKey, cL1Vkh, cL2Vkh])
  const wrap1 = DatumUtils.mkConstr(0, [constrKey])

  // Nested map structure: { "" => { "" => amount } }
  const emptyBytes = DatumUtils.mkBytes('')
  const mapVal = CardanoWASM.PlutusMapValues.new()
  mapVal.add(DatumUtils.mkInt(amount))
  const innerMap = DatumUtils.mkMap([[emptyBytes, mapVal]])

  const outerMapVal = CardanoWASM.PlutusMapValues.new()
  outerMapVal.add(innerMap)
  const outerMap = DatumUtils.mkMap([[emptyBytes, outerMapVal]])

  // Final datum structure
  return DatumUtils.mkConstr(0, [wrap1, outerMap])
}

// Usage example từ playground
const datum = buildDatum(
  'ee91e90e791e4cd983d1b1f331d1e8eb',
  '326cd6bff6114c4d14ebf2385883aac43c4e64476e6a47314f9b2003',
  'f602ad4b16ec2e1a96989dc140eacf546359695cfece8510c8d1c0ac',
  '4000000'
)

console.log('Complex Datum:', datum.to_json(CardanoWASM.PlutusDatumSchema.DetailedSchema))
```

### Slot Time Conversion

```typescript
import { TimeUtils, SLOT_CONFIG_NETWORK } from '@hydra-sdk/core'

// Convert giữa slot và timestamp (utility functions thực tế)
const slotTimeUtils = {
  // Slot → Unix timestamp
  slotToTime: (slot: number) => {
    return TimeUtils.slotToBeginUnixTime(slot, SLOT_CONFIG_NETWORK.PREPROD)
  },
  
  // Unix timestamp → Slot
  timeToSlot: (timestamp: number) => {
    return TimeUtils.unixTimeToEnclosingSlot(timestamp, SLOT_CONFIG_NETWORK.PREPROD)
  },
  
  // Current slot
  getCurrentSlot: () => {
    return TimeUtils.unixTimeToEnclosingSlot(Date.now(), SLOT_CONFIG_NETWORK.PREPROD)
  },
  
  // Future slot calculation
  getFutureSlot: (hoursFromNow: number) => {
    const futureTime = Date.now() + (hoursFromNow * 60 * 60 * 1000)
    return TimeUtils.unixTimeToEnclosingSlot(futureTime, SLOT_CONFIG_NETWORK.PREPROD)
  },
  
  // Human readable time from slot (pattern từ playground)
  slotToReadableTime: (slot: number) => {
    const timestamp = TimeUtils.slotToBeginUnixTime(slot, SLOT_CONFIG_NETWORK.PREPROD)
    return new Date(timestamp).toLocaleString('vi-VN')
  }
}

// Demo usage
const currentSlot = slotTimeUtils.getCurrentSlot()
const futureSlot = slotTimeUtils.getFutureSlot(24) // 24 giờ từ bây giờ

console.log(`Current slot: ${currentSlot}`)
console.log(`Current time: ${slotTimeUtils.slotToReadableTime(currentSlot)}`)
console.log(`Future slot (+24h): ${futureSlot}`)
console.log(`Future time: ${slotTimeUtils.slotToReadableTime(futureSlot)}`)

// Example từ playground: slot 102672789
console.log('Historical slot 102672789:', slotTimeUtils.slotToReadableTime(102672789))
```

### Transaction Timing

```typescript
import { TimeUtils, SLOT_CONFIG_NETWORK } from '@hydra-sdk/core'

// Tính toán validity range cho transaction
const calculateTxValidityRange = (network: 'MAINNET' | 'PREPROD' | 'PREVIEW') => {
  const now = Date.now()
  const slotConfig = SLOT_CONFIG_NETWORK[network]
  const currentSlot = TimeUtils.unixTimeToEnclosingSlot(now, slotConfig)
  
  // Valid từ slot hiện tại
  const validFrom = currentSlot
  
  // Valid đến 2 giờ từ bây giờ
  const twoHoursFromNow = now + (2 * 60 * 60 * 1000)
  const validUntil = TimeUtils.unixTimeToEnclosingSlot(twoHoursFromNow, slotConfig)
  
  return { validFrom, validUntil }
}
```

## Ví Dụ Provider

### Provider Fallback System

```typescript
import { ProviderUtils } from '@hydra-sdk/core'

// Tạo hệ thống provider mạnh mẽ với fallbacks
class RobustProvider {
  private providers: any[]
  
  constructor() {
    this.providers = [
      new ProviderUtils.BlockfrostProvider({
        apiKey: 'primary-project-id',
        network: 'preprod'
      }),
      new ProviderUtils.BlockfrostProvider({
        apiKey: 'backup-project-id',
        network: 'preprod'
      }),
      new ProviderUtils.OgmiosProvider({
        network: 'preprod',
        apiEndpoint: 'http://localhost:1337'
      })
    ]
  }
  
  async getUtxos(address: string) {
    for (let i = 0; i < this.providers.length; i++) {
      try {
        console.log(`Thử provider ${i + 1}...`)
        const utxos = await this.providers[i].fetcher.fetchAddressUTxOs(address)
        console.log(`Thành công với provider ${i + 1}`)
        return utxos
      } catch (error) {
        console.warn(`Provider ${i + 1} thất bại:`, error.message)
        if (i === this.providers.length - 1) {
          throw new Error('Tất cả providers đều thất bại')
        }
      }
    }
  }
  
  async getProtocolParameters() {
    for (let i = 0; i < this.providers.length; i++) {
      try {
        return await this.providers[i].getProtocolParameters()
      } catch (error) {
        if (i === this.providers.length - 1) {
          throw error
        }
      }
    }
  }
}

// Sử dụng
const robustProvider = new RobustProvider()

try {
  const utxos = await robustProvider.getUtxos('addr1qx2fxv2umyhttkxyxp8x0dlpdt3k6cwng5pxj3jhsydzer')
  console.log('Lấy được UTxOs:', utxos.length)
} catch (error) {
  console.error('Thất bại lấy UTxOs:', error.message)
}
```

```typescript
      if (typeof value === 'number' && !isNaN(value)) {

      } else {

      }

    } catch (error) {

      return null
    }
  }
```
