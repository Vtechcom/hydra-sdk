# Ví dụ Tích Hợp Hydra

Ví dụ kết nối tới Hydra Node/Head, quản lý vòng đời Head, và xử lý giao dịch.

## Quản Lý Hydra Head Cơ Bản

```typescript
import { HydraBridge } from '@hydra-sdk/bridge'

async function manageHydraHead() {
  const bridge = new HydraBridge({
    url: 'ws://localhost:4001',
    verbose: true
  })

  // Thiết lập các event handler trước khi kết nối
  bridge.events.on('onConnected', () => {
    console.log('Đã kết nối! Đang khởi tạo Head...')
    bridge.commands.init()
  })

  bridge.events.on('onMessage', (payload) => {
    switch (payload.tag) {
      case 'HeadIsInitializing':
        console.log('Head đang khởi tạo với', payload.parties.length, 'bên tham gia')
        break
      case 'HeadIsOpen':
        console.log('Head đã mở và sẵn sàng cho giao dịch!')
        break
      case 'TxValid':
        console.log('Giao dịch được xác nhận:', payload.transaction.txId)
        break
      case 'HeadIsClosed':
        console.log('Head đã đóng, đang thực hiện fanout...')
        bridge.commands.fanout()
        break
      case 'HeadIsFinalized':
        console.log('Vòng đời Head hoàn tất')
        bridge.disconnect()
        break
    }
  })

  // Kết nối để bắt đầu quá trình
  await bridge.connect()
}
```

## Xử Lý Giao Dịch trong Hydra Head

```typescript
import { HydraBridge } from '@hydra-sdk/bridge'
import { TxBuilder } from '@hydra-sdk/transaction'
import { AppWallet, deserializeTx } from '@hydra-sdk/core'

async function processHydraTransactions() {
  const bridge = new HydraBridge({
    url: 'ws://localhost:4001',
    verbose: true
  })

  const wallet = new AppWallet({
    networkId: 0,
    key: {
      type: 'mnemonic',
      words: AppWallet.brew()
    }
  })

  bridge.events.on('onConnected', async () => {
    // Lấy protocol parameters cho transaction building
    const protocolParams = await bridge.getProtocolParameters()
    
    // Tạo transaction builder được cấu hình cho Hydra
    const txBuilder = new TxBuilder({
      isHydra: true,
      params: protocolParams
    })

    // Lấy thông tin account và UTxOs
    const account = wallet.getAccount(0, 0)
    const addressUtxos = await bridge.queryAddressUTxO(account.baseAddressBech32)

    if (addressUtxos.length === 0) {
      console.log('Không tìm thấy UTxOs cho địa chỉ')
      return
    }

    // Xây dựng và ký giao dịch
    const tx = await txBuilder
      .setInputs(addressUtxos)
      .addLovelaceOutput(account.baseAddressBech32, '1000000')
      .setChangeAddress(account.baseAddressBech32)
      .complete()

    const txId = deserializeTx(tx.to_hex()).transaction_hash().to_hex()
    const signedTx = await wallet.signTx(tx.to_hex(), false, 0, 0)

    // Gửi lên Hydra Head
    const result = await bridge.submitTxSync({
      type: 'Witnessed Tx ConwayEra',
      description: 'Ledger Cddl Format',
      cborHex: signedTx,
      txId
    }, { timeout: 30000 })

    console.log('Giao dịch thành công:', result)
  })

  await bridge.connect()
}
```

```typescript
  })

}
```

## Vòng Đời Head Đầy Đủ với Xử Lý Lỗi

```typescript
async function fullHeadLifecycleWithErrorHandling() {
  const bridge = new HydraBridge({
    url: 'ws://localhost:4001',
    verbose: true
  })

  let headState = 'idle'
  let reconnectAttempts = 0
  const maxReconnectAttempts = 5

  // Xử lý lỗi kết nối
  bridge.events.on('onConnectError', async (error) => {
    console.error('Kết nối thất bại:', error)
    
    if (reconnectAttempts < maxReconnectAttempts) {
      reconnectAttempts++
      console.log(`Đang thử kết nối lại (${reconnectAttempts}/${maxReconnectAttempts})...`)
      
      await new Promise(resolve => setTimeout(resolve, 5000))
      await bridge.connect()
    } else {
      console.error('Đã đạt số lần thử kết nối lại tối đa')
    }
  })

  bridge.events.on('onConnected', () => {
    console.log('Đã kết nối, đặt lại bộ đếm kết nối lại')
    reconnectAttempts = 0
    
    if (headState === 'idle') {
      console.log('Bắt đầu khởi tạo head...')
      bridge.commands.init()
    }
  })

  bridge.events.on('onMessage', async (payload) => {
    switch (payload.tag) {
      case 'Greetings':
        headState = payload.headStatus.toLowerCase()
        console.log('Trạng thái head hiện tại:', headState)
        break
        
      case 'HeadIsInitializing':
        headState = 'initializing'
        console.log('Head đang khởi tạo với các bên:', payload.parties.length)
        break
        
      case 'HeadIsOpen':
        headState = 'open'
        const utxoCount = Object.keys(payload.utxo).length
        console.log(`Head đã mở với ${utxoCount} UTxOs`)
        
        // Head sẵn sàng cho giao dịch
        break
        
      case 'HeadIsClosed':
        headState = 'closed'
        console.log('Head đã đóng, đang thực hiện fanout...')
        bridge.commands.fanout()
        break
        
      case 'HeadIsFinalized':
        headState = 'finalized'
        console.log('Vòng đời Head hoàn tất')
        await bridge.disconnect()
        break
        
      case 'TxInvalid':
        console.error('Giao dịch không hợp lệ:', {
          transaction: payload.transaction,
          reason: payload.validationError
        })
        break
        
      case 'CommandFailed':
        console.error('Lệnh thất bại:', payload.clientInput)
        break
        
      case 'InvalidInput':
        console.error('Input không hợp lệ:', payload.reason)
        break
    }
  })

  await bridge.connect()
}
```

## Ví dụ Tích Hợp Vue.js

```vue
<script lang="ts" setup>
import { ref, onMounted, onBeforeUnmount } from 'vue'
import { HydraBridge, HexcoreConnector } from '@hydra-sdk/bridge'

const hydraBridge = ref<HydraBridge | null>(null)
const isConnected = ref(false)
const headStatus = ref('disconnected')
const messages = ref<string[]>([])

const addMessage = (message: string) => {
  messages.value.unshift(`${new Date().toLocaleTimeString()}: ${message}`)
  if (messages.value.length > 10) {
    messages.value = messages.value.slice(0, 10)
  }
}

onMounted(async () => {
  hydraBridge.value = new HydraBridge({
    url: 'ws://localhost:4001',
    verbose: true
  })

  hydraBridge.value.events.on('onConnected', () => {
    isConnected.value = true
    addMessage('Đã kết nối tới Hydra Bridge')
  })

  hydraBridge.value.events.on('onDisconnected', () => {
    isConnected.value = false
    addMessage('Đã ngắt kết nối khỏi Hydra Bridge')
  })

  hydraBridge.value.events.on('onMessage', (payload) => {
    headStatus.value = payload.tag
    addMessage(`Nhận được: ${payload.tag}`)
  })

  await hydraBridge.value.connect()
})

onBeforeUnmount(() => {
  if (hydraBridge.value) {
    hydraBridge.value.disconnect()
  }
})

const initHead = () => {
  if (hydraBridge.value && isConnected.value) {
    hydraBridge.value.commands.init()
    addMessage('Đã gửi lệnh init')
  }
}

const closeHead = () => {
  if (hydraBridge.value && isConnected.value) {
    hydraBridge.value.commands.close()
    addMessage('Đã gửi lệnh close')
  }
}
```

## Operations Commit và Decommit

```typescript
import { HydraBridge } from '@hydra-sdk/bridge'
import { TxBuilder } from '@hydra-sdk/transaction'
import { AppWallet } from '@hydra-sdk/core'

async function commitAndDecommitExample() {
  const bridge = new HydraBridge({
    url: 'ws://localhost:4001',
    verbose: true
  })

  const wallet = new AppWallet({
    networkId: 0,
    key: { type: 'mnemonic', words: AppWallet.brew() }
  })

  bridge.events.on('onMessage', async (payload) => {
    switch (payload.tag) {
      case 'HeadIsOpen':
        console.log('Head đã mở, có thể commit UTxOs')
        
        // Ví dụ: Commit UTxOs vào head
        const commitTxCbor = 'your_commit_transaction_cbor'
        const commitResult = await bridge.commit({
          cborHex: commitTxCbor,
          description: 'Commit ban đầu'
        })
        console.log('Kết quả commit:', commitResult)
        break
        
      case 'Committed':
        console.log('UTxO đã được commit thành công:', payload.utxo)
        
        // Sau đó, bạn có thể decommit UTxOs
        const decommitTxCbor = 'your_decommit_transaction_cbor'
        const decommitTxId = 'decommit_transaction_id'
        
        const decommitResult = await bridge.commands.decommit({
          cborHex: decommitTxCbor,
          txId: decommitTxId,
          timeout: 60000
        })
        console.log('Kết quả decommit:', decommitResult)
        break
        
      case 'DecommitApproved':
        console.log('Decommit đã được phê duyệt:', payload.decommitTxId)
        break
        
      case 'DecommitFinalized':
        console.log('Decommit đã hoàn tất, UTxOs có sẵn trên L1')
        break
    }
  })

  await bridge.connect()
}
```

```typescript
    }
  })
```

## Best Practices

1. **Thiết Lập Event**: Luôn thiết lập event handlers trước khi gọi `connect()`
2. **Xử Lý Lỗi**: Triển khai xử lý lỗi toàn diện và logic reconnection
3. **Quản Lý State**: Theo dõi trạng thái Head để đảm bảo các operation hợp lệ
4. **Authentication**: Sử dụng JWT token phù hợp cho kết nối Hexcore
5. **Cleanup Tài Nguyên**: Luôn gọi `disconnect()` khi hoàn tất
6. **Validation Giao Dịch**: Validate giao dịch trước khi submit
7. **Cấu Hình Timeout**: Đặt timeout phù hợp cho các operation
8. **Logging**: Bật verbose mode để debug trong quá trình phát triển
