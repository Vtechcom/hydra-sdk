## Vue.js Example

Wallet creation in a Vue 3 application:

```vue
<script setup lang="ts">
import { ref, reactive } from 'vue'
import { AppWallet, generateMnemonic, NETWORK_ID } from '@hydra-sdk/core'

interface WalletInfo {
  mnemonic: string
  baseAddress: string
  stakeAddress: string
  paymentKeyHash: string
}

const walletInfo = ref<WalletInfo | null>(null)
const loading = ref(false)
const mnemonicInput = ref('')
const networkId = ref(NETWORK_ID.TESTNET)

const createNewWallet = async () => {
  loading.value = true
  
  try {
    // Generate new mnemonic
    const mnemonic = generateMnemonic()
    
    // Create wallet
    const wallet = new AppWallet({
      key: {
        type: 'mnemonic',
        words: mnemonic.split(' ')
      },
      networkId: networkId.value
    })

    const account = wallet.getAccount(0, 0)
    
    walletInfo.value = {
      mnemonic,
      baseAddress: account.baseAddressBech32,
      stakeAddress: account.stakeAddressBech32,
      paymentKeyHash: account.paymentKeyHash
    }
  } catch (error) {
    console.error('Error creating wallet:', error)
  } finally {
    loading.value = false
  }
}

const restoreWallet = async () => {
  if (!mnemonicInput.value.trim()) {
    alert('Please enter a valid mnemonic phrase')
    return
  }

  loading.value = true
  
  try {
    const wallet = new AppWallet({
      key: {
        type: 'mnemonic',
        words: mnemonicInput.value.trim().split(' ')
      },
      networkId: networkId.value
    })

    const account = wallet.getAccount(0, 0)
    
    walletInfo.value = {
      mnemonic: mnemonicInput.value.trim(),
      baseAddress: account.baseAddressBech32,
      stakeAddress: account.stakeAddressBech32,
      paymentKeyHash: account.paymentKeyHash
    }
  } catch (error) {
    console.error('Error restoring wallet:', error)
    alert('Error restoring wallet. Please check your mnemonic phrase.')
  } finally {
    loading.value = false
  }
}

const clearWallet = () => {
  walletInfo.value = null
  mnemonicInput.value = ''
}
</script>
```

## React Example

Wallet creation in a React application:

```tsx
// WalletCreation.tsx
import React, { useState } from 'react';
import { AppWallet, generateMnemonic, NETWORK_ID } from '@hydra-sdk/core';

interface WalletInfo {
  mnemonic: string;
  baseAddress: string;
  stakeAddress: string;
  paymentKeyHash: string;
}

const WalletCreation: React.FC = () => {
  const [walletInfo, setWalletInfo] = useState<WalletInfo | null>(null);
  const [loading, setLoading] = useState(false);
  const [mnemonicInput, setMnemonicInput] = useState('');
  const [networkId, setNetworkId] = useState(NETWORK_ID.TESTNET);

  const createNewWallet = async () => {
    setLoading(true);
    
    try {
      // Generate new mnemonic
      const mnemonic = generateMnemonic();
      
      // Create wallet
      const wallet = new AppWallet({
        key: {
          type: 'mnemonic',
          words: mnemonic.split(' ')
        },
        networkId
      });

      const account = wallet.getAccount(0, 0);
      
      setWalletInfo({
        mnemonic,
        baseAddress: account.baseAddressBech32,
        stakeAddress: account.stakeAddressBech32,
        paymentKeyHash: account.paymentKeyHash
      });
    } catch (error) {
      console.error('Error creating wallet:', error);
    } finally {
      setLoading(false);
    }
  };

  const restoreWallet = async () => {
    if (!mnemonicInput.trim()) {
      alert('Please enter a valid mnemonic phrase');
      return;
    }

    setLoading(true);
    
    try {
      const wallet = new AppWallet({
        key: {
          type: 'mnemonic',
          words: mnemonicInput.trim().split(' ')
        },
        networkId
      });

      const account = wallet.getAccount(0, 0);
      
      setWalletInfo({
        mnemonic: mnemonicInput.trim(),
        baseAddress: account.baseAddressBech32,
        stakeAddress: account.stakeAddressBech32,
        paymentKeyHash: account.paymentKeyHash
      });
    } catch (error) {
      console.error('Error restoring wallet:', error);
      alert('Error restoring wallet. Please check your mnemonic phrase.');
    } finally {
      setLoading(false);
    }
  };

  const clearWallet = () => {
    setWalletInfo(null);
    setMnemonicInput('');
  };

  return (
    <div className="wallet-container">
      <h1>Hydra Wallet Creation</h1>
      
      <div className="controls">
        <div className="network-selection">
          <label>
            <input 
              type="radio"
              checked={networkId === NETWORK_ID.TESTNET}
              onChange={() => setNetworkId(NETWORK_ID.TESTNET)}
            />
            Testnet
          </label>
          <label>
            <input 
              type="radio"
              checked={networkId === NETWORK_ID.MAINNET}
              onChange={() => setNetworkId(NETWORK_ID.MAINNET)}
            />
            Mainnet
          </label>
        </div>

        <button 
          onClick={createNewWallet} 
          disabled={loading}
          className="btn btn-primary"
        >
          {loading ? 'Creating...' : 'Create New Wallet'}
        </button>

        <div className="restore-section">
          <textarea
            value={mnemonicInput}
            onChange={(e) => setMnemonicInput(e.target.value)}
            placeholder="Enter your 24-word mnemonic phrase to restore wallet"
            rows={3}
            className="mnemonic-input"
          />
          <button 
            onClick={restoreWallet} 
            disabled={loading}
            className="btn btn-secondary"
          >
            Restore Wallet
          </button>
        </div>

        {walletInfo && (
          <button 
            onClick={clearWallet}
            className="btn btn-danger"
          >
            Clear Wallet
          </button>
        )}
      </div>

      {walletInfo && (
        <div className="wallet-info">
          <h3>Wallet Information</h3>
          <div className="info-item">
            <label>Mnemonic Phrase:</label>
            <code>{walletInfo.mnemonic}</code>
          </div>
          <div className="info-item">
            <label>Base Address:</label>
            <code>{walletInfo.baseAddress}</code>
          </div>
          <div className="info-item">
            <label>Stake Address:</label>
            <code>{walletInfo.stakeAddress}</code>
          </div>
          <div className="info-item">
            <label>Payment Key Hash:</label>
            <code>{walletInfo.paymentKeyHash}</code>
          </div>
        </div>
      )}
    </div>
  );
};

export default WalletCreation;
```

Wallet creation in a Svelte application:

```svelte
<!-- WalletCreation.svelte -->
<script lang="ts">
  import { onMount } from 'svelte';
  import { writable } from 'svelte/store';
  import { AppWallet, generateMnemonic, NETWORK_ID } from '@hydra-sdk/core';

  interface WalletInfo {
    mnemonic: string;
    baseAddress: string;
    stakeAddress: string;
    paymentKeyHash: string;
  }

  let walletInfo: WalletInfo | null = null;
  let loading = false;
  let mnemonicInput = '';
  let networkId = NETWORK_ID.TESTNET;

  // Svelte store for wallet state
  const walletStore = writable<WalletInfo | null>(null);

  onMount(() => {
    // Load wallet from localStorage
    const stored = localStorage.getItem('hydra-wallet');
    if (stored) {
      walletInfo = JSON.parse(stored);
      walletStore.set(walletInfo);
    }
  });

  // Subscribe to wallet store changes
  walletStore.subscribe(value => {
    if (value) {
      localStorage.setItem('hydra-wallet', JSON.stringify(value));
    }
  });

  async function createNewWallet() {
    loading = true;
    
    try {
      const mnemonic = generateMnemonic();
      
      const wallet = new AppWallet({
        key: {
          type: 'mnemonic',
          words: mnemonic.split(' ')
        },
        networkId
      });

      const account = wallet.getAccount(0, 0);
      
      walletInfo = {
        mnemonic,
        baseAddress: account.baseAddressBech32,
        stakeAddress: account.stakeAddressBech32,
        paymentKeyHash: account.paymentKeyHash
      };

      walletStore.set(walletInfo);
    } catch (error) {
      console.error('Error creating wallet:', error);
      alert('Failed to create wallet');
    } finally {
      loading = false;
    }
  }

  async function restoreWallet() {
    if (!mnemonicInput.trim()) {
      alert('Please enter a valid mnemonic phrase');
      return;
    }

    loading = true;
    
    try {
      const wallet = new AppWallet({
        key: {
          type: 'mnemonic',
          words: mnemonicInput.trim().split(' ')
        },
        networkId
      });

      const account = wallet.getAccount(0, 0);
      
      walletInfo = {
        mnemonic: mnemonicInput.trim(),
        baseAddress: account.baseAddressBech32,
        stakeAddress: account.stakeAddressBech32,
        paymentKeyHash: account.paymentKeyHash
      };

      walletStore.set(walletInfo);
    } catch (error) {
      console.error('Error restoring wallet:', error);
      alert('Error restoring wallet. Please check your mnemonic phrase.');
    } finally {
      loading = false;
    }
  }

  function clearWallet() {
    walletInfo = null;
    mnemonicInput = '';
    localStorage.removeItem('hydra-wallet');
    walletStore.set(null);
  }
</script>
```

## Key Features

All examples demonstrate:

- **Mnemonic Generation**: Creating new 24-word mnemonic phrases
- **Wallet Restoration**: Restoring wallets from existing mnemonics  
- **Network Selection**: Choosing between Testnet and Mainnet
- **Address Generation**: Getting base and stake addresses
- **Key Management**: Accessing payment and stake key hashes
- **Persistence**: Saving wallet data to localStorage (where applicable)
- **Error Handling**: Proper error management and user feedback
- **Loading States**: UI feedback during async operations

Each example is framework-specific and follows the respective framework's best practices for state management, lifecycle handling, and component architecture.
